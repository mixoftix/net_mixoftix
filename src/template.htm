FUNCTION compress_public_key(X, Y):
    parity = Y % 2
    IF parity == 1 THEN
        suffix = '1'
    ELSE
        suffix = '2'
    END IF
    compressed_key = X + '*' + suffix
    RETURN compressed_key
END FUNCTION
